def compose(
  t : Type,
  u : Type,
  v : Type,
  f : +U[ω] (t -> -F u),
  g : +U[ω] (u -> -F v),
  x : t
) -> -F v {
  run y <- f(x);
  become g(y)
}

def captured(
  a : Type,
  d : Type,
  u : +U[ω] (a -> -F d),
  h : +U[ω] (d -> -F d)
) -> -F (+U[ω] (a -> -F d)) {
  return thunk { become compose(a, d, d, u, h) }
}
